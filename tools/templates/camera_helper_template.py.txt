"""
CameraHelper Engine:
Controls viewport camera navigation, locking orbit pivot to (0,0,0) or object bounding boxes.
Disables erratic mouse tumble, provides ViewCube presets (iso, top, front, etc.),
gradual zoom, zoom-to-fit controls, and built-in smooth cinematic glide transitions.
"""

from typing import Optional, Tuple, Dict, Any, Union
from pxr import Usd, UsdGeom, Sdf, Gf, Tf

try:
    import carb.settings
except ImportError:
    carb = None

try:
    import omni.kit.viewport.utility as vp_utils
except ImportError:
    vp_utils = None


VIEWCUBE_DIRECTIONS = {
    "iso":    Gf.Vec3d(1.0, 1.0, 1.0).GetNormalized(),
    "top":    Gf.Vec3d(0.0001, 1.0, 0.0001).GetNormalized(),
    "bottom": Gf.Vec3d(0.0001, -1.0, 0.0001).GetNormalized(),
    "front":  Gf.Vec3d(0.0, 0.0, 1.0).GetNormalized(),
    "back":   Gf.Vec3d(0.0, 0.2, -1.0).GetNormalized(),
    "left":   Gf.Vec3d(-1.0, 0.2, 0.0).GetNormalized(),
    "right":  Gf.Vec3d(1.0, 0.2, 0.0).GetNormalized(),
}


class CameraHelper:
    """Orbit Pivot, ViewCube Presets, Zoom Navigation, and Smooth Glide Engine."""

    def __init__(self, enable_smooth_glide: bool = True, default_glide_duration: float = 1.0):
        self.camera_path: str = "/OmniverseKit_Persp"
        self._current_pivot: Gf.Vec3d = Gf.Vec3d(0.0, 0.0, 0.0)
        self._current_distance: float = 1000.0
        self._current_view_direction: Gf.Vec3d = VIEWCUBE_DIRECTIONS["iso"]

        # Smooth Glide Interpolator configuration
        self.enable_smooth_glide: bool = enable_smooth_glide
        self.default_glide_duration: float = default_glide_duration
        self._is_gliding: bool = False
        self._glide_elapsed: float = 0.0
        self._glide_duration: float = 1.0
        self._glide_start_pos: Gf.Vec3d = Gf.Vec3d(0.0, 0.0, 0.0)
        self._glide_start_pivot: Gf.Vec3d = Gf.Vec3d(0.0, 0.0, 0.0)
        self._glide_start_distance: float = 1000.0
        self._glide_target_pos: Gf.Vec3d = Gf.Vec3d(0.0, 0.0, 0.0)
        self._glide_target_pivot: Gf.Vec3d = Gf.Vec3d(0.0, 0.0, 0.0)
        self._glide_target_distance: float = 1000.0

        # Default overview tracking for clean selection & unselection resets
        self._default_pivot: Gf.Vec3d = Gf.Vec3d(0.0, 0.0, 0.0)
        self._default_distance: float = 1000.0
        self._was_zoomed: bool = False
        self._pre_zoom_dist: Optional[float] = None

        self._disable_cursor_tumble()

    def set_camera_path(self, cam_path: str) -> None:
        """Sets the dedicated camera prim path for this level (e.g. /OmniverseKit_level_01_Cam)."""
        if cam_path:
            self.camera_path = str(cam_path)

    def bind_viewport_camera(self, stage: Usd.Stage, cam_path: Optional[str] = None) -> None:
        """
        Cleanly binds the active viewport to this camera prim, flushing stale manipulator state.
        Matches the proven complete-data-centre technique (HallService / RegionService).
        """
        target_path = cam_path or self.camera_path or "/OmniverseKit_Persp"
        if vp_utils:
            try:
                v_api = vp_utils.get_active_viewport()
                if v_api:
                    if str(v_api.camera_path) == str(target_path):
                        v_api.camera_path = "/OmniverseKit_Persp"
                    v_api.camera_path = target_path
            except Exception:
                pass
        try:
            import omni.kit.commands
            omni.kit.commands.execute("SetViewportCamera", camera_path=target_path)
        except Exception:
            pass

    def _disable_cursor_tumble(self) -> None:
        """
        Disables automatic Kit orbit-framing-on-click so custom pivot locks operate cleanly.
        Matches complete-data-centre exactly, cleaning up any corrupted persistent settings.
        """
        if carb:
            try:
                settings = carb.settings.get_settings()
                if settings:
                    settings.set("/exts/omni.kit.viewport.navigation.camera_manipulator/orbitAutoFrameSelected", False)
                    # Clean up any bad keys that might have been persistently written
                    for key in [
                        "/app/viewport/autoFrame/mode",
                        "/persistent/app/viewport/autoFrame/mode",
                        "/app/viewport/camera/tumbleAtCursor",
                        "/persistent/app/viewport/camera/tumbleAtCursor",
                        "/app/viewport/manipulator/camera/tumbleAtCursor",
                        "/persistent/app/viewport/manipulator/camera/tumbleAtCursor",
                        "/persistent/exts/omni.kit.manipulator.camera/objectCentric/type",
                        "/exts/omni.kit.manipulator.camera/objectCentric/type",
                        "/persistent/exts/omni.kit.manipulator.camera/objectCentric/useGroundPlane",
                        "/exts/omni.kit.manipulator.camera/objectCentric/useGroundPlane",
                    ]:
                        try:
                            settings.destroy_item(key)
                        except Exception:
                            pass
            except Exception:
                pass

    def get_active_camera_prim(self, stage: Usd.Stage) -> Optional[Usd.Prim]:
        """Resolves active viewport camera prim, defining target path if missing."""
        if not stage:
            return None

        cam_path = None
        if vp_utils:
            try:
                v_api = vp_utils.get_active_viewport()
                if v_api and hasattr(v_api, "camera_path") and v_api.camera_path:
                    cam_path = str(v_api.camera_path)
            except Exception:
                pass

        if not cam_path:
            cam_path = self.camera_path or "/OmniverseKit_Persp"

        prim = stage.GetPrimAtPath(cam_path)
        if not prim or not prim.IsValid():
            cam = UsdGeom.Camera.Define(stage, cam_path)
            prim = cam.GetPrim()

        return prim

    def _configure_camera_prim(self, p: Usd.Prim, cam_mat: Gf.Matrix4d, dist: float, focal_length: float = 20.0) -> None:
        """
        Configures transform, perspective attributes, centerOfInterest, and unlocks camera.
        Strictly follows complete-data-centre pattern to ensure Kit manipulator operates stably.
        """
        try:
            xformable = UsdGeom.Xformable(p)
            xform_op = None
            for op in xformable.GetOrderedXformOps():
                if op.GetOpType() == UsdGeom.XformOp.TypeTransform:
                    xform_op = op
                    break
            if not xform_op:
                try:
                    xform_op = xformable.AddTransformOp()
                except Exception:
                    attr = p.GetAttribute("xformOp:transform")
                    if attr and attr.IsValid():
                        xform_op = UsdGeom.XformOp(attr)
            if xform_op:
                xform_op.Set(cam_mat)

            cam_api = UsdGeom.Camera(p)
            cam_api.CreateProjectionAttr().Set(UsdGeom.Tokens.perspective)
            cam_api.CreateFocalLengthAttr().Set(float(focal_length))
            cam_api.CreateFocusDistanceAttr().Set(float(dist))
            cam_api.CreateClippingRangeAttr().Set(Gf.Vec2f(1.0, 1000000.0))

            # CRITICAL: In Omniverse Kit, orbit pivot, pan speed, and zoom scale are governed by
            # omni:kit:centerOfInterest of type Vector3d(0, 0, -dist) in camera local space!
            coi_attr = p.GetAttribute("omni:kit:centerOfInterest")
            if not coi_attr or not coi_attr.IsValid():
                coi_attr = p.CreateAttribute(
                    "omni:kit:centerOfInterest", Sdf.ValueTypeNames.Vector3d, True, Sdf.VariabilityUniform
                )
            coi_attr.Set(Gf.Vec3d(0, 0, -float(dist)))

            # Unlock camera prim so user can orbit & pan freely
            for lock_name in ["camera:omni:kit:cameraLock", "omni:kit:cameraLock"]:
                l_attr = p.GetAttribute(lock_name)
                if l_attr and l_attr.IsValid():
                    l_attr.Set(False)

        except Exception as e:
            print(f"[CameraHelper] Note during camera prim configuration {p.GetPath()}: {e}")

    def _apply_pose(
        self,
        stage: Usd.Stage,
        cam_pos: Gf.Vec3d,
        look_at: Gf.Vec3d,
        distance: float,
        focal_length: float = 20.0,
        cam_path: Optional[str] = None
    ) -> None:
        """
        Computes lookAt matrix and updates camera prim in SessionLayer and RootLayer.
        Dispatches TransformPrimCommand and binds active viewport cleanly.
        """
        if not stage:
            return

        target_cam_path = cam_path or self.camera_path or "/OmniverseKit_Persp"

        forward = (look_at - cam_pos).GetNormalized()
        up_candidate = Gf.Vec3d(0.0, 1.0, 0.0)

        # Handle singularity when looking directly along Y axis
        if abs(forward[1]) > 0.999:
            up_candidate = Gf.Vec3d(0.0, 0.0, -1.0 if forward[1] > 0 else 1.0)

        cam_mat = Gf.Matrix4d()
        cam_mat.SetLookAt(cam_pos, look_at, up_candidate)
        cam_mat = cam_mat.GetInverse()

        cams_to_update = [target_cam_path]
        if target_cam_path != "/OmniverseKit_Persp":
            cams_to_update.append("/OmniverseKit_Persp")

        # 1. Update on SessionLayer (CRITICAL: overrides viewport interactive pan/zoom opinions)
        session_layer = stage.GetSessionLayer()
        if session_layer:
            with Usd.EditContext(stage, Usd.EditTarget(session_layer)):
                for c_path in cams_to_update:
                    p = stage.GetPrimAtPath(c_path)
                    if not p or not p.IsValid():
                        if c_path != "/OmniverseKit_Persp":
                            cam = UsdGeom.Camera.Define(stage, c_path)
                            p = cam.GetPrim()
                    if p and p.IsValid():
                        self._configure_camera_prim(p, cam_mat, distance, focal_length)

        # 2. Update on RootLayer
        root_layer = stage.GetRootLayer()
        if root_layer and root_layer != session_layer:
            with Usd.EditContext(stage, Usd.EditTarget(root_layer)):
                for c_path in cams_to_update:
                    p = stage.GetPrimAtPath(c_path)
                    if not p or not p.IsValid():
                        if c_path != "/OmniverseKit_Persp":
                            cam = UsdGeom.Camera.Define(stage, c_path)
                            p = cam.GetPrim()
                    if p and p.IsValid():
                        self._configure_camera_prim(p, cam_mat, distance, focal_length)

        # 3. Notify Omniverse Kit Viewport Command Dispatcher
        try:
            import omni.kit.commands
            omni.kit.commands.execute(
                "TransformPrimCommand",
                path=target_cam_path,
                new_transform_matrix=cam_mat,
            )
        except Exception:
            pass

        # 4. Switch / sync active viewport to target camera cleanly
        if vp_utils:
            try:
                v_api = vp_utils.get_active_viewport()
                if v_api:
                    if str(v_api.camera_path) != str(target_cam_path):
                        v_api.camera_path = target_cam_path
            except Exception:
                pass

        try:
            import omni.kit.commands
            omni.kit.commands.execute("SetViewportCamera", camera_path=target_cam_path)
        except Exception:
            pass

        self._current_distance = distance
        self._current_pivot = look_at
        vec = cam_pos - look_at
        if vec.GetLength() > 1.0:
            self._current_view_direction = vec.GetNormalized()

    def _start_glide(
        self,
        stage: Usd.Stage,
        target_pos: Gf.Vec3d,
        target_pivot: Gf.Vec3d,
        target_distance: float,
        duration: Optional[float] = None
    ) -> None:
        """Initiates a smooth interpolated glide toward target camera transform."""
        dur = duration if duration is not None else self.default_glide_duration
        if dur <= 0.001 or not self.enable_smooth_glide:
            self._is_gliding = False
            self._apply_pose(stage, target_pos, target_pivot, target_distance)
            return

        # Query current camera position to ensure smooth takeoff without jumping
        cur_pos, cur_dist, _ = self.get_current_camera_pose(stage)

        self._glide_start_pos = cur_pos
        self._glide_start_pivot = self._current_pivot
        self._glide_start_distance = cur_dist

        self._glide_target_pos = target_pos
        self._glide_target_pivot = target_pivot
        self._glide_target_distance = target_distance

        self._glide_duration = max(0.05, dur)
        self._glide_elapsed = 0.0
        self._is_gliding = True

    def on_update(self, dt: float, stage: Optional[Usd.Stage] = None) -> None:
        """
        Frame update step for camera glide interpolation.
        Uses a quintic smootherstep curve for zero acceleration and deceleration jerk.
        """
        if not self._is_gliding or not stage:
            return

        self._glide_elapsed += dt
        t = min(1.0, self._glide_elapsed / self._glide_duration)

        # Quintic Smootherstep: 6t^5 - 15t^4 + 10t^3
        ease_t = t * t * t * (t * (t * 6.0 - 15.0) + 10.0)

        cur_pos = self._glide_start_pos + (self._glide_target_pos - self._glide_start_pos) * ease_t
        cur_piv = self._glide_start_pivot + (self._glide_target_pivot - self._glide_start_pivot) * ease_t
        cur_dist = self._glide_start_distance + (self._glide_target_distance - self._glide_start_distance) * ease_t

        self._apply_pose(stage, cur_pos, cur_piv, cur_dist)

        if t >= 1.0:
            self._is_gliding = False

    def get_current_camera_pose(self, stage: Usd.Stage) -> Tuple[Gf.Vec3d, float, Gf.Vec3d]:
        """
        Returns (camera_world_pos, distance_to_pivot, view_direction) based on active camera prim in USD stage.
        Falls back to internal tracked state if camera prim cannot be resolved.
        """
        cam_prim = self.get_active_camera_prim(stage)
        if cam_prim and cam_prim.IsValid():
            try:
                xformable = UsdGeom.Xformable(cam_prim)
                world_xform = xformable.ComputeLocalToWorldTransform(Usd.TimeCode.Default())
                cam_pos = world_xform.ExtractTranslation()
                vec = cam_pos - self._current_pivot
                dist = vec.GetLength()
                if dist > 1.0:
                    view_dir = vec.GetNormalized()
                    return cam_pos, dist, view_dir
            except Exception:
                pass
        return (
            self._current_pivot + (self._current_view_direction * self._current_distance),
            self._current_distance,
            self._current_view_direction
        )

    def set_camera_pose(
        self,
        stage: Usd.Stage,
        position: Union[Tuple[float, float, float], Gf.Vec3d] = (0.0, 0.0, 1000.0),
        rotation: Optional[Union[Tuple[float, float, float], Gf.Vec3d]] = None,
        distance: float = 1000.0,
        pivot_point: Union[Tuple[float, float, float], Gf.Vec3d] = (0.0, 0.0, 0.0),
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> None:
        """
        Sets explicit camera position (x, y, z) and locks the orbit pivot strictly to pivot_point.
        """
        pos_vec = Gf.Vec3d(float(position[0]), float(position[1]), float(position[2]))
        piv_vec = Gf.Vec3d(float(pivot_point[0]), float(pivot_point[1]), float(pivot_point[2]))
        dist = (pos_vec - piv_vec).GetLength()
        if dist < 1.0:
            dist = float(distance)

        use_smooth = self.enable_smooth_glide if smooth is None else smooth
        if use_smooth:
            self._start_glide(stage, pos_vec, piv_vec, dist, duration=duration)
        else:
            self._is_gliding = False
            self._apply_pose(stage, cam_pos=pos_vec, look_at=piv_vec, distance=dist)

    def set_camera_pivot(
        self,
        stage: Usd.Stage,
        pivot_point: Union[Tuple[float, float, float], Gf.Vec3d] = (0.0, 0.0, 0.0),
        distance: float = 1000.0,
        view: Optional[str] = None,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> None:
        """
        Sets the camera orbit pivot point and positions the camera at the given distance.
        Optionally switches to view preset (e.g. 'iso', 'front', 'top') if view is specified.
        """
        if view and view.lower() in VIEWCUBE_DIRECTIONS:
            self._current_view_direction = VIEWCUBE_DIRECTIONS[view.lower()]

        target_pivot = Gf.Vec3d(pivot_point[0], pivot_point[1], pivot_point[2])
        target_dist = max(1.0, float(distance))
        target_cam_pos = target_pivot + (self._current_view_direction * target_dist)

        # Record default level overview pivot and distance
        self._default_pivot = target_pivot
        self._default_distance = target_dist
        self._was_zoomed = False

        use_smooth = self.enable_smooth_glide if smooth is None else smooth
        if use_smooth:
            self._start_glide(stage, target_cam_pos, target_pivot, target_dist, duration=duration)
        else:
            self._is_gliding = False
            self._apply_pose(stage, target_cam_pos, target_pivot, target_dist)

    def set_camera_pivot_to_object(
        self,
        stage: Usd.Stage,
        prim_path: str,
        zoom_distance: Optional[float] = None,
        distance: Optional[float] = None,
        zoom: Optional[bool] = None,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> bool:
        """
        Moves orbit pivot to the geometric center (midpoint) of the target object's bounding box.
        - If zoom_distance <= 0 (or zoom is False):
          Camera stays stationary (cam_pos does NOT translate/zoom).
          Only orbit pivot and centerOfInterest are updated to the object.
        - If zoom_distance > 0:
          Smooth glides camera to the specified zoom distance focusing on the object.
        """
        if not stage or not prim_path:
            return False

        prim = stage.GetPrimAtPath(prim_path)
        if not prim or not prim.IsValid():
            print(f"[CameraHelper] Cannot focus: prim '{prim_path}' is not valid on stage.")
            return False

        bbox_cache = UsdGeom.BBoxCache(Usd.TimeCode.Default(), [
            UsdGeom.Tokens.default_, UsdGeom.Tokens.render, UsdGeom.Tokens.proxy
        ])
        bound = bbox_cache.ComputeWorldBound(prim)
        rng = bound.ComputeAlignedRange()
        if not rng.IsEmpty():
            mid = rng.GetMidpoint()
        else:
            try:
                xformable = UsdGeom.Xformable(prim)
                mid = xformable.ComputeLocalToWorldTransform(Usd.TimeCode.Default()).ExtractTranslation()
            except Exception:
                print(f"[CameraHelper] Bounding box is empty for '{prim_path}'")
                return False

        cam_pos, cur_dist, cur_dir = self.get_current_camera_pose(stage)

        # Resolve effective zoom distance:
        # Priority: zoom_distance argument, then distance if zoom is True/None
        effective_zoom_dist = 0.0
        if zoom_distance is not None:
            effective_zoom_dist = float(zoom_distance)
        elif zoom is False:
            effective_zoom_dist = 0.0
        elif distance is not None:
            effective_zoom_dist = float(distance)
        elif zoom is True:
            size = (rng.GetMax() - rng.GetMin()).GetLength() if not rng.IsEmpty() else 100.0
            effective_zoom_dist = max(100.0, size * 1.5)

        if effective_zoom_dist <= 0.0:
            # 🎯 MODE 1: ZOOM = 0 -> KAMERA DIAM (TANPA ZOOMING)
            # Posisi kamera (cam_pos) tidak berubah sama sekali!
            self._was_zoomed = False
            self._is_gliding = False

            vec_to_cam = cam_pos - mid
            actual_dist = max(1.0, vec_to_cam.GetLength())
            self._current_pivot = mid
            self._current_distance = actual_dist
            self._current_view_direction = vec_to_cam.GetNormalized()

            # Terapkan pose: cam_pos tetap di tempatnya, arahkan look_at ke objek, centerOfInterest = actual_dist
            self._apply_pose(stage, cam_pos=cam_pos, look_at=mid, distance=actual_dist)
            print(f"[CameraHelper] Pivot locked to '{prim_path}' at {mid} (Stationary, distance={actual_dist:.1f})")
            return True
        else:
            # 🎯 MODE 2: ZOOM > 0 -> FOKUS & SMOOTH ZOOM
            if not self._was_zoomed:
                self._pre_zoom_dist = cur_dist

            self._was_zoomed = True
            target_dist = max(10.0, effective_zoom_dist)

            vec_to_cam = cam_pos - mid
            view_dir = vec_to_cam.GetNormalized() if vec_to_cam.GetLength() > 1.0 else cur_dir
            target_cam_pos = mid + (view_dir * target_dist)

            use_smooth = self.enable_smooth_glide if smooth is None else smooth
            if use_smooth:
                self._start_glide(stage, target_cam_pos, mid, target_dist, duration=duration)
            else:
                self._is_gliding = False
                self._apply_pose(stage, target_cam_pos, mid, target_dist)

            print(f"[CameraHelper] Zooming to '{prim_path}' at distance {target_dist:.1f} (Pivot: {mid})")
            return True

    def reset_camera_pivot(
        self,
        stage: Usd.Stage,
        distance: Optional[float] = None,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> None:
        """
        Resets orbit pivot back to default level origin (0, 0, 0).
        - If previous selection was stationary (zoom == 0):
          Camera stays stationary at current position, only pivot & centerOfInterest return to origin.
        - If previous selection had zoom > 0:
          Smooth glides back to original overview distance at origin.
        """
        if not stage:
            return

        cam_pos, cur_dist, cur_dir = self.get_current_camera_pose(stage)
        target_pivot = self._default_pivot

        if not self._was_zoomed and distance is None:
            # 🎯 KONDISI 1: Tadi zoom == 0 -> KAMERA TETAP DIAM
            # Tidak berpindah koordinat, hanya pivot kembali ke origin
            vec_to_cam = cam_pos - target_pivot
            actual_dist = max(1.0, vec_to_cam.GetLength())

            self._is_gliding = False
            self._current_pivot = target_pivot
            self._current_distance = actual_dist
            self._current_view_direction = vec_to_cam.GetNormalized()

            self._apply_pose(stage, cam_pos=cam_pos, look_at=target_pivot, distance=actual_dist)
            print(f"[CameraHelper] Reset pivot to {target_pivot} (Stationary, distance={actual_dist:.1f})")
        else:
            # 🎯 KONDISI 2: Tadi zoom > 0 -> KEMBALI KE OVERVIEW
            target_dist = float(distance) if distance is not None else (self._pre_zoom_dist or self._default_distance or 1000.0)
            target_dist = max(10.0, target_dist)

            vec_to_cam = cam_pos - target_pivot
            view_dir = vec_to_cam.GetNormalized() if vec_to_cam.GetLength() > 1.0 else cur_dir
            target_cam_pos = target_pivot + (view_dir * target_dist)

            use_smooth = self.enable_smooth_glide if smooth is None else smooth
            if use_smooth:
                self._start_glide(stage, target_cam_pos, target_pivot, target_dist, duration=duration)
            else:
                self._is_gliding = False
                self._apply_pose(stage, target_cam_pos, target_pivot, target_dist)

            self._was_zoomed = False
            self._pre_zoom_dist = None
            print(f"[CameraHelper] Smooth reset to overview distance {target_dist:.1f} (Pivot: {target_pivot})")

    def set_camera_view(
        self,
        stage: Usd.Stage,
        view_name: str = "iso",
        distance: Optional[float] = None,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> bool:
        """
        Rotates camera to standard dice presets ('iso', 'top', 'front', 'back', 'left', 'right', 'bottom').
        Preserves current pivot point, or updates distance if provided.
        """
        key = (view_name or "iso").strip().lower()
        if key not in VIEWCUBE_DIRECTIONS:
            return False

        if distance is not None:
            self._current_distance = max(1.0, float(distance))
            self._default_distance = self._current_distance
        self._was_zoomed = False

        self._current_view_direction = VIEWCUBE_DIRECTIONS[key]
        target_cam_pos = self._current_pivot + (self._current_view_direction * self._current_distance)

        use_smooth = self.enable_smooth_glide if smooth is None else smooth
        if use_smooth:
            self._start_glide(stage, target_cam_pos, self._current_pivot, self._current_distance, duration=duration)
        else:
            self._is_gliding = False
            self._apply_pose(stage, target_cam_pos, self._current_pivot, self._current_distance)
        return True

    def camera_zoom_in(
        self,
        stage: Usd.Stage,
        factor: float = 0.8,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> None:
        """Zooms camera in toward the active pivot point."""
        new_dist = max(10.0, self._current_distance * factor)
        self.set_camera_distance(stage, new_dist, smooth=smooth, duration=duration)

    def camera_zoom_out(
        self,
        stage: Usd.Stage,
        factor: float = 1.25,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> None:
        """Zooms camera out away from the active pivot point."""
        new_dist = max(10.0, self._current_distance * factor)
        self.set_camera_distance(stage, new_dist, smooth=smooth, duration=duration)

    def set_camera_distance(
        self,
        stage: Usd.Stage,
        distance: float,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> None:
        """Sets an exact distance from the camera to the active pivot."""
        target_dist = max(10.0, float(distance))
        target_cam_pos = self._current_pivot + (self._current_view_direction * target_dist)

        use_smooth = self.enable_smooth_glide if smooth is None else smooth
        if use_smooth:
            self._start_glide(stage, target_cam_pos, self._current_pivot, target_dist, duration=duration)
        else:
            self._is_gliding = False
            self._apply_pose(stage, target_cam_pos, self._current_pivot, target_dist)

    def zoom_to_fit_object(
        self,
        stage: Usd.Stage,
        prim_path: str,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> bool:
        """Fits camera framing snugly around the target object's bounding box."""
        if not stage or not prim_path:
            return False

        prim = stage.GetPrimAtPath(prim_path)
        if not prim or not prim.IsValid():
            return False

        bbox_cache = UsdGeom.BBoxCache(Usd.TimeCode.Default(), [UsdGeom.Tokens.default_])
        bound = bbox_cache.ComputeWorldBound(prim)
        rng = bound.ComputeAlignedRange()
        if rng.IsEmpty():
            return False

        mid = rng.GetMidpoint()
        size = (rng.GetMax() - rng.GetMin()).GetLength()
        optimal_dist = max(50.0, size * 1.8)
        self.set_camera_pivot(stage, pivot_point=mid, distance=optimal_dist, smooth=smooth, duration=duration)
        return True
