"""
CameraHelper Engine:
Controls viewport camera navigation, locking orbit pivot to (0,0,0) or object bounding boxes.
Disables erratic mouse tumble, provides ViewCube presets (iso, top, front, etc.),
and provides gradual zoom and zoom-to-fit controls.
"""

from typing import Optional, Tuple, Dict, Any, Union
from pxr import Usd, UsdGeom, Sdf, Gf

try:
    import carb.settings
except ImportError:
    carb = None

try:
    import omni.kit.viewport.utility as vp_utils
except ImportError:
    vp_utils = None


VIEWCUBE_DIRECTIONS = {
    "iso":    Gf.Vec3d(1.0, 1.0, 1.0).GetNormalized(),
    "top":    Gf.Vec3d(0.0001, 1.0, 0.0001).GetNormalized(),
    "bottom": Gf.Vec3d(0.0001, -1.0, 0.0001).GetNormalized(),
    "front":  Gf.Vec3d(0.0, 0.0, 1.0).GetNormalized(),
    "back":   Gf.Vec3d(0.0, 0.2, -1.0).GetNormalized(),
    "left":   Gf.Vec3d(-1.0, 0.2, 0.0).GetNormalized(),
    "right":  Gf.Vec3d(1.0, 0.2, 0.0).GetNormalized(),
}


class CameraHelper:
    """Orbit Pivot, ViewCube Presets, and Zoom Navigation Engine."""

    def __init__(self):
        self._current_pivot: Gf.Vec3d = Gf.Vec3d(0.0, 0.0, 0.0)
        self._current_distance: float = 1000.0
        self._current_view_direction: Gf.Vec3d = VIEWCUBE_DIRECTIONS["iso"]
        self._disable_cursor_tumble()

    def _disable_cursor_tumble(self) -> None:
        """Disables erratic Kit tumble-at-cursor behavior to maintain stable orbit pivots."""
        if carb:
            try:
                settings = carb.settings.get_settings()
                if settings:
                    settings.set("/app/viewport/camera/tumbleAtCursor", False)
                    settings.set("/persistent/app/viewport/camera/tumbleAtCursor", False)
                    settings.set("/app/viewport/manipulator/camera/tumbleAtCursor", False)
                    settings.set("/persistent/app/viewport/manipulator/camera/tumbleAtCursor", False)
                    settings.set("/persistent/exts/omni.kit.manipulator.camera/objectCentric/type", 0)
                    settings.set("/exts/omni.kit.manipulator.camera/objectCentric/type", 0)
                    settings.set("/persistent/exts/omni.kit.manipulator.camera/objectCentric/useGroundPlane", False)
                    settings.set("/exts/omni.kit.manipulator.camera/objectCentric/useGroundPlane", False)
                    settings.set("/exts/omni.kit.viewport.navigation.camera_manipulator/orbitAutoFrameSelected", False)
                    settings.set("/persistent/exts/omni.kit.viewport.navigation.camera_manipulator/orbitAutoFrameSelected", False)
                    settings.set("/app/viewport/autoFrame/mode", "disabled")
                    settings.set("/persistent/app/viewport/autoFrame/mode", "disabled")
            except Exception:
                pass

    def get_active_camera_prim(self, stage: Usd.Stage) -> Optional[Usd.Prim]:
        """Resolves active viewport camera prim, defining /OmniverseKit_Persp if missing."""
        if not stage:
            return None

        cam_path = None
        if vp_utils:
            try:
                v_api = vp_utils.get_active_viewport()
                if v_api and hasattr(v_api, "camera_path") and v_api.camera_path:
                    cam_path = str(v_api.camera_path)
            except Exception:
                pass

        if not cam_path:
            cam_path = "/OmniverseKit_Persp"

        prim = stage.GetPrimAtPath(cam_path)
        if not prim or not prim.IsValid():
            cam = UsdGeom.Camera.Define(stage, cam_path)
            prim = cam.GetPrim()

        return prim

    def _configure_camera_prim(self, p: Usd.Prim, cam_mat: Gf.Matrix4d, dist: float, focal_length: float = 20.0) -> None:
        """
        Configures transform, perspective attributes, centerOfInterest, and unlocks camera.
        Explicitly clears conflicting individual translate/rotate ops to ensure transform op takes precedence.
        """
        try:
            xformable = UsdGeom.Xformable(p)

            # Check if xformOp:transform already exists on the prim
            trans_attr = p.GetAttribute("xformOp:transform")
            order_attr = xformable.GetXformOpOrderAttr()
            cur_order = list(order_attr.Get()) if order_attr and order_attr.IsValid() and order_attr.Get() else []

            if "xformOp:transform" in cur_order and trans_attr and trans_attr.IsValid():
                trans_attr.Set(cam_mat)
            else:
                # Clear individual translate/rotate/scale ops that override transform
                for op_name in ["xformOp:translate", "xformOp:rotateXYZ", "xformOp:scale"]:
                    attr = p.GetAttribute(op_name)
                    if attr and attr.IsValid():
                        attr.Clear()

                if "xformOp:transform" not in cur_order:
                    xformable.ClearXformOpOrder()
                    op = xformable.AddTransformOp()
                    op.Set(cam_mat)
                else:
                    if not trans_attr or not trans_attr.IsValid():
                        trans_attr = p.CreateAttribute("xformOp:transform", Sdf.ValueTypeNames.Matrix4d)
                    trans_attr.Set(cam_mat)

            cam_api = UsdGeom.Camera(p)
            cam_api.CreateProjectionAttr().Set(UsdGeom.Tokens.perspective)
            cam_api.CreateFocusDistanceAttr().Set(dist)
            cam_api.CreateFocalLengthAttr().Set(focal_length)
            cam_api.CreateClippingRangeAttr().Set(Gf.Vec2f(1.0, 1000000.0))

            # Set Center of Interest: In camera space looking along -Z, COI is (0, 0, -dist)
            coi_attr = p.GetAttribute("omni:kit:centerOfInterest")
            if not coi_attr or not coi_attr.IsValid():
                coi_attr = p.CreateAttribute(
                    "omni:kit:centerOfInterest",
                    Sdf.ValueTypeNames.Vector3d,
                    True,
                    Sdf.VariabilityUniform
                )
            coi_attr.Set(Gf.Vec3d(0.0, 0.0, -dist))

            for lock_name in ["camera:omni:kit:cameraLock", "omni:kit:cameraLock"]:
                l_attr = p.GetAttribute(lock_name)
                if l_attr and l_attr.IsValid():
                    l_attr.Set(False)

        except Exception as e:
            print(f"[CameraHelper] Error configuring camera prim: {e}")

    def _apply_pose(self, stage: Usd.Stage, cam_pos: Gf.Vec3d, look_at: Gf.Vec3d, distance: float, focal_length: float = 20.0) -> None:
        """Calculates LookAt transform and synchronizes camera prim across Session and Root layers."""
        if not stage:
            return

        self._disable_cursor_tumble()

        up = Gf.Vec3d(0.0, 1.0, 0.0)
        view_dir = (look_at - cam_pos).GetNormalized()
        if abs(Gf.Dot(view_dir, up)) > 0.999:
            up = Gf.Vec3d(0.0, 0.0, -1.0) if view_dir[1] > 0 else Gf.Vec3d(0.0, 0.0, 1.0)

        rot_mat = Gf.Matrix4d().SetLookAt(cam_pos, look_at, up)
        cam_mat = rot_mat.GetInverse()

        cam_prim = self.get_active_camera_prim(stage)
        if not cam_prim or not cam_prim.IsValid():
            print("[CameraHelper] Warning: No valid camera prim found on stage.")
            return

        cam_path = cam_prim.GetPath().pathString

        # Apply to both SessionLayer and RootLayer for total stability across stages
        session_layer = stage.GetSessionLayer()
        if session_layer:
            with Usd.EditContext(stage, Usd.EditTarget(session_layer)):
                p = stage.GetPrimAtPath(cam_path)
                if not p or not p.IsValid():
                    cam = UsdGeom.Camera.Define(stage, cam_path)
                    p = cam.GetPrim()
                if p and p.IsValid():
                    self._configure_camera_prim(p, cam_mat, distance, focal_length)

        root_layer = stage.GetRootLayer()
        if root_layer and root_layer != session_layer:
            with Usd.EditContext(stage, Usd.EditTarget(root_layer)):
                p = stage.GetPrimAtPath(cam_path)
                if not p or not p.IsValid():
                    cam = UsdGeom.Camera.Define(stage, cam_path)
                    p = cam.GetPrim()
                if p and p.IsValid():
                    self._configure_camera_prim(p, cam_mat, distance, focal_length)

        # Sync active viewport
        if vp_utils:
            try:
                v_api = vp_utils.get_active_viewport()
                if v_api:
                    v_api.camera_path = cam_path
            except Exception:
                pass

        try:
            import omni.kit.commands
            omni.kit.commands.execute("SetViewportCamera", camera_path=cam_path)
        except Exception:
            pass

        # Update internal tracking
        self._current_distance = distance
        self._current_pivot = look_at

        print(f"[CameraHelper] Camera posed: Pos=({cam_pos[0]:.1f}, {cam_pos[1]:.1f}, {cam_pos[2]:.1f}) -> Orbit Pivot=({look_at[0]:.1f}, {look_at[1]:.1f}, {look_at[2]:.1f}) | Dist={distance:.1f}")

    def set_camera_pose(
        self,
        stage: Usd.Stage,
        position: Union[Tuple[float, float, float], Gf.Vec3d] = (0.0, 0.0, 1000.0),
        rotation: Optional[Union[Tuple[float, float, float], Gf.Vec3d]] = None,
        distance: float = 1000.0,
        pivot_point: Union[Tuple[float, float, float], Gf.Vec3d] = (0.0, 0.0, 0.0)
    ) -> None:
        """
        Sets explicit camera position (x, y, z) and locks the orbit pivot strictly to pivot_point (default: 0, 0, 0).
        """
        pos_vec = Gf.Vec3d(float(position[0]), float(position[1]), float(position[2]))
        piv_vec = Gf.Vec3d(float(pivot_point[0]), float(pivot_point[1]), float(pivot_point[2]))
        dist = (pos_vec - piv_vec).GetLength()
        if dist < 1.0:
            dist = float(distance)

        print(f"[CameraHelper] Setting camera pose to Pos={pos_vec}, Orbit Pivot locked to {piv_vec}")
        self._apply_pose(stage, cam_pos=pos_vec, look_at=piv_vec, distance=dist)

    def set_camera_pivot(
        self,
        stage: Usd.Stage,
        pivot_point: Union[Tuple[float, float, float], Gf.Vec3d] = (0.0, 0.0, 0.0),
        distance: float = 1000.0,
        view: Optional[str] = None
    ) -> None:
        """
        Sets the camera orbit pivot point and positions the camera at the given distance.
        Optionally switches to view preset (e.g. 'iso', 'front', 'top') if view is specified.
        """
        if view and view.lower() in VIEWCUBE_DIRECTIONS:
            self._current_view_direction = VIEWCUBE_DIRECTIONS[view.lower()]

        self._current_pivot = Gf.Vec3d(pivot_point[0], pivot_point[1], pivot_point[2])
        self._current_distance = max(1.0, float(distance))

        cam_pos = self._current_pivot + (self._current_view_direction * self._current_distance)
        self._apply_pose(stage, cam_pos, self._current_pivot, self._current_distance)

    def get_current_camera_pose(self, stage: Usd.Stage) -> Tuple[Gf.Vec3d, float, Gf.Vec3d]:
        """
        Returns (camera_world_pos, distance_to_pivot, view_direction) based on active camera prim in USD stage.
        Falls back to internal tracked state if camera prim cannot be resolved.
        """
        cam_prim = self.get_active_camera_prim(stage)
        if cam_prim and cam_prim.IsValid():
            try:
                xformable = UsdGeom.Xformable(cam_prim)
                world_xform = xformable.ComputeLocalToWorldTransform(Usd.TimeCode.Default())
                cam_pos = world_xform.ExtractTranslation()
                vec = cam_pos - self._current_pivot
                dist = vec.GetLength()
                if dist > 1.0:
                    view_dir = vec.GetNormalized()
                    return cam_pos, dist, view_dir
            except Exception:
                pass
        return (
            self._current_pivot + (self._current_view_direction * self._current_distance),
            self._current_distance,
            self._current_view_direction
        )

    def set_camera_pivot_to_object(
        self,
        stage: Usd.Stage,
        prim_path: str,
        distance: Optional[float] = None,
        zoom: bool = True
    ) -> bool:
        """
        Moves orbit pivot to the geometric center (midpoint) of the target object's bounding box.

        Args:
            stage: Active USD stage.
            prim_path: Path of the target prim.
            distance: Explicit camera distance. If specified as float (e.g. 500.0), camera zooms to this distance.
            zoom: If False, preserves the current camera distance without any zoom change.
                  If True and distance is None, automatically calculates distance based on bounding box.
        """
        if not stage or not prim_path:
            return False

        prim = stage.GetPrimAtPath(prim_path)
        if not prim or not prim.IsValid():
            print(f"[CameraHelper] Cannot focus: prim '{prim_path}' is not valid on stage.")
            return False

        bbox_cache = UsdGeom.BBoxCache(Usd.TimeCode.Default(), [
            UsdGeom.Tokens.default_, UsdGeom.Tokens.render, UsdGeom.Tokens.proxy
        ])
        bound = bbox_cache.ComputeWorldBound(prim)
        rng = bound.ComputeAlignedRange()
        if rng.IsEmpty():
            print(f"[CameraHelper] Bounding box is empty for '{prim_path}'")
            return False

        mid = rng.GetMidpoint()
        size = (rng.GetMax() - rng.GetMin()).GetLength()

        # Query current camera orientation so we preserve viewing angle smoothly
        _, cur_dist, cur_dir = self.get_current_camera_pose(stage)
        self._current_view_direction = cur_dir

        if not zoom:
            # Preserve current camera zoom distance
            target_dist = cur_dist
            print(f"[CameraHelper] Locking orbit pivot to '{prim_path}' at Center=({mid[0]:.1f}, {mid[1]:.1f}, {mid[2]:.1f}) (no zoom, preserving dist={target_dist:.1f})")
        elif distance is not None:
            # Custom zoom distance specified
            target_dist = max(10.0, float(distance))
            print(f"[CameraHelper] Locking orbit pivot to '{prim_path}' at Center=({mid[0]:.1f}, {mid[1]:.1f}, {mid[2]:.1f}) (custom zoom dist={target_dist:.1f})")
        else:
            # Auto-fit based on bounding box size
            target_dist = max(100.0, size * 1.5)
            print(f"[CameraHelper] Locking orbit pivot to '{prim_path}' at Center=({mid[0]:.1f}, {mid[1]:.1f}, {mid[2]:.1f}) (auto-fit dist={target_dist:.1f})")

        self.set_camera_pivot(stage, pivot_point=mid, distance=target_dist)
        return True

    def reset_camera_pivot(self, stage: Usd.Stage) -> None:
        """Resets camera pivot back to the global origin (0, 0, 0)."""
        _, cur_dist, cur_dir = self.get_current_camera_pose(stage)
        self._current_view_direction = cur_dir
        print(f"[CameraHelper] Resetting orbit pivot to origin (0.0, 0.0, 0.0) | preserving dist={cur_dist:.1f}")
        self.set_camera_pivot(stage, pivot_point=(0.0, 0.0, 0.0), distance=cur_dist)

    def set_camera_view(self, stage: Usd.Stage, view_name: str = "iso", distance: Optional[float] = None) -> bool:
        """
        Rotates camera to standard dice presets ('iso', 'top', 'front', 'back', 'left', 'right', 'bottom').
        Preserves current pivot point, or updates distance if provided.
        """
        key = (view_name or "iso").strip().lower()
        if key not in VIEWCUBE_DIRECTIONS:
            return False

        if distance is not None:
            self._current_distance = max(1.0, float(distance))

        self._current_view_direction = VIEWCUBE_DIRECTIONS[key]
        cam_pos = self._current_pivot + (self._current_view_direction * self._current_distance)
        print(f"[CameraHelper] Set view preset '{key.upper()}': Pivot={self._current_pivot}, Distance={self._current_distance:.1f}")
        self._apply_pose(stage, cam_pos, self._current_pivot, self._current_distance)
        return True

    def camera_zoom_in(self, stage: Usd.Stage, factor: float = 0.8) -> None:
        """Zooms camera in toward the active pivot point."""
        self._current_distance = max(10.0, self._current_distance * factor)
        cam_pos = self._current_pivot + (self._current_view_direction * self._current_distance)
        self._apply_pose(stage, cam_pos, self._current_pivot, self._current_distance)

    def camera_zoom_out(self, stage: Usd.Stage, factor: float = 1.25) -> None:
        """Zooms camera out away from the active pivot point."""
        self._current_distance = max(10.0, self._current_distance * factor)
        cam_pos = self._current_pivot + (self._current_view_direction * self._current_distance)
        self._apply_pose(stage, cam_pos, self._current_pivot, self._current_distance)

    def set_camera_distance(self, stage: Usd.Stage, distance: float) -> None:
        """Sets an exact distance from the camera to the active pivot."""
        self._current_distance = max(10.0, float(distance))
        cam_pos = self._current_pivot + (self._current_view_direction * self._current_distance)
        self._apply_pose(stage, cam_pos, self._current_pivot, self._current_distance)

    def zoom_to_fit_object(self, stage: Usd.Stage, prim_path: str) -> bool:
        """Fits camera framing snugly around the target object's bounding box."""
        if not stage or not prim_path:
            return False

        prim = stage.GetPrimAtPath(prim_path)
        if not prim or not prim.IsValid():
            return False

        bbox_cache = UsdGeom.BBoxCache(Usd.TimeCode.Default(), [UsdGeom.Tokens.default_])
        bound = bbox_cache.ComputeWorldBound(prim)
        rng = bound.ComputeAlignedRange()
        if rng.IsEmpty():
            return False

        mid = rng.GetMidpoint()
        size = (rng.GetMax() - rng.GetMin()).GetLength()
        optimal_dist = max(50.0, size * 1.8)
        self.set_camera_pivot(stage, pivot_point=mid, distance=optimal_dist)
        return True
