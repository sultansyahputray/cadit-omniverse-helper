"""
BaseLevelHandler Abstract Contract:
Inherits and exposes all four core helper engines:
1. VisibilityHelper (hide_objects, dynamic cutaways, restore_all_visibility)
2. ProjectionHelper (frustum math, screen positions, dirty check)
3. ColorHelper (unopinionated color injection, restore_all_colors)
4. CameraHelper (orbit pivot locked to 0,0,0, ViewCube presets, zoom)
"""

from abc import ABC, abstractmethod
from typing import Dict, Any, Optional
from pxr import Usd

from .context import NavigationContext
from .helpers.visibility_helper import VisibilityHelper
from .helpers.projection_helper import ProjectionHelper
from .helpers.color_helper import ColorHelper
from .helpers.camera_helper import CameraHelper


class BaseLevelHandler(ABC):
    """Abstract Base Class for modular level handlers."""

    level_id: str = "base_level"
    display_name: str = "Base Level"

    # [OPTIONAL] USD map stage path if this level uses a distinct USD file:
    map_url: Optional[str] = None

    # [OPTIONAL] Root prim path for this level:
    root_prim_path: str = "/World"

    # [OPTIONAL] Prims tracked for 2D Web Dashboard projection:
    tracked_objects: Dict[str, str] = {}

    # [OPTIONAL] Object selection click zoom distance:
    # - 0.0 : Stationary camera (NO zoom, camera stays still, orbit pivot locked to object)
    # - > 0.0 (e.g. 350.0) : Smooth glide camera to the specified zoom distance
    click_zoom_distance: float = 0.0

    def __init__(self, manager):
        self.manager = manager
        self.visibility_helper = VisibilityHelper()
        self.projection_helper = ProjectionHelper()
        self.color_helper = ColorHelper()
        self.camera_helper = CameraHelper()
        self.camera_path: str = f"/OmniverseKit_{self.level_id}_Cam"
        self.camera_helper.set_camera_path(self.camera_path)

    def bind_viewport_camera(self, stage: Usd.Stage) -> None:
        """Cleanly binds active viewport to this level's camera prim and flushes manipulator state."""
        self.camera_helper.bind_viewport_camera(stage, self.camera_path)

    # --- Delegated Visibility Methods ---
    def hide_objects(self, stage: Usd.Stage, prim_paths: list) -> None:
        self.visibility_helper.hide_objects(stage, prim_paths)

    def show_objects(self, stage: Usd.Stage, prim_paths: list) -> None:
        self.visibility_helper.show_objects(stage, prim_paths)

    def register_cutaway_walls(self, walls_config: dict) -> None:
        self.visibility_helper.register_cutaway_walls(walls_config)

    def update_perspective_cutaway(self, stage: Usd.Stage, cam_pos=None, center=(0.0, 0.0, 0.0), dot_threshold=0.15) -> None:
        self.visibility_helper.update_perspective_cutaway(stage, cam_pos, center, dot_threshold)

    def restore_all_visibility(self, stage: Usd.Stage) -> None:
        self.visibility_helper.restore_all_visibility(stage)

    # --- Delegated Projection Methods ---
    def track_object(
        self,
        key: str,
        prim_path: str,
        compute_bounds: bool = False,
        anchor_point: str = "top_center"
    ) -> None:
        """
        Registers a USD prim to track screen coordinates (x%, y%, and optional width%).
        Args:
            key: Unique identifier in telemetry output (e.g. 'Rack_01', 'Chiller_A')
            prim_path: USD stage prim path (e.g. '/World/DataCentre/Rack_01')
            compute_bounds: If True, calculates projected bounding width for dynamic zoom scaling
            anchor_point: 'top_center', 'center', or 'bottom_center'
        """
        self.tracked_objects[key] = {
            "path": prim_path,
            "bounds": compute_bounds,
            "anchor": anchor_point
        }

    def untrack_object(self, key: str) -> None:
        """Removes a USD prim from tracked objects and clears its cached projection."""
        self.tracked_objects.pop(key, None)
        if hasattr(self, "projection_helper"):
            self.projection_helper.clear_object_cache(key)

    def get_tracked_screen_positions(self, stage: Usd.Stage, client_aspect=None) -> Dict[str, Dict[str, Any]]:
        return self.projection_helper.compute_screen_positions(stage, self.tracked_objects, client_aspect)

    def clear_projection_cache(self) -> None:
        self.projection_helper.clear_projection_cache()

    # --- Delegated Color, Glow & Pulse Methods ---
    def set_color(
        self,
        stage: Usd.Stage,
        prim_path: str,
        color: Any,
        glow: bool = False,
        glow_intensity: float = 3000.0,
        pulse: bool = False,
        pulse_speed: float = 1.0
    ) -> bool:
        return self.color_helper.set_color(
            stage, prim_path, color, glow=glow, glow_intensity=glow_intensity, pulse=pulse, pulse_speed=pulse_speed
        )

    def set_glow(
        self,
        stage: Usd.Stage,
        prim_path: str,
        color: Any,
        intensity: float = 3000.0,
        pulse: bool = False,
        pulse_speed: float = 1.0
    ) -> bool:
        return self.color_helper.set_glow(
            stage, prim_path, color, intensity=intensity, pulse=pulse, pulse_speed=pulse_speed
        )

    def set_colors(
        self,
        stage: Usd.Stage,
        mapping: dict,
        glow: bool = False,
        glow_intensity: float = 3000.0
    ) -> None:
        self.color_helper.set_colors(stage, mapping, glow=glow, glow_intensity=glow_intensity)

    def restore_color(self, stage: Usd.Stage, prim_path: str) -> None:
        self.color_helper.restore_color(stage, prim_path)

    def restore_all_colors(self, stage: Usd.Stage) -> None:
        self.color_helper.restore_all_colors(stage)

    # --- Delegated Camera Methods (with built-in smooth glide) ---
    def set_camera_pose(
        self,
        stage: Usd.Stage,
        position=(0.0, 0.0, 1000.0),
        rotation=(0.0, 0.0, 0.0),
        distance=1000.0,
        pivot_point=(0.0, 0.0, 0.0),
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> None:
        self.camera_helper.set_camera_pose(
            stage, position, rotation, distance, pivot_point=pivot_point, smooth=smooth, duration=duration
        )

    def set_camera_pivot(
        self,
        stage: Usd.Stage,
        pivot_point=(0.0, 0.0, 0.0),
        distance=1000.0,
        view: Optional[str] = None,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> None:
        self.camera_helper.set_camera_pivot(
            stage, pivot_point, distance, view=view, smooth=smooth, duration=duration
        )

    def set_camera_pivot_to_object(
        self,
        stage: Usd.Stage,
        prim_path: str,
        zoom_distance: Optional[float] = None,
        distance: Optional[float] = None,
        zoom: Optional[bool] = None,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> bool:
        return self.camera_helper.set_camera_pivot_to_object(
            stage, prim_path, zoom_distance=zoom_distance, distance=distance, zoom=zoom, smooth=smooth, duration=duration
        )

    def reset_camera_pivot(
        self,
        stage: Usd.Stage,
        distance: Optional[float] = None,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> None:
        self.camera_helper.reset_camera_pivot(stage, distance=distance, smooth=smooth, duration=duration)

    def set_camera_view(
        self,
        stage: Usd.Stage,
        view_name: str = "iso",
        distance: Optional[float] = None,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> bool:
        return self.camera_helper.set_camera_view(
            stage, view_name, distance=distance, smooth=smooth, duration=duration
        )

    def camera_zoom_in(
        self,
        stage: Usd.Stage,
        factor=0.8,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> None:
        self.camera_helper.camera_zoom_in(stage, factor, smooth=smooth, duration=duration)

    def camera_zoom_out(
        self,
        stage: Usd.Stage,
        factor=1.25,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> None:
        self.camera_helper.camera_zoom_out(stage, factor, smooth=smooth, duration=duration)

    def zoom_to_fit_object(
        self,
        stage: Usd.Stage,
        prim_path: str,
        smooth: Optional[bool] = None,
        duration: Optional[float] = None
    ) -> bool:
        return self.camera_helper.zoom_to_fit_object(stage, prim_path, smooth=smooth, duration=duration)

    # --- Delegated Selection Methods ---
    def get_selected_prim(self) -> Optional[str]:
        """Returns the currently active single selected prim path."""
        if self.manager and hasattr(self.manager, "selection_helper") and self.manager.selection_helper:
            return self.manager.selection_helper.get_selected_prim()
        return None

    def clear_selection(self) -> None:
        """Clears all viewport selection."""
        if self.manager and hasattr(self.manager, "selection_helper") and self.manager.selection_helper:
            self.manager.selection_helper.clear_selection()

    def set_selection(self, prim_path: str) -> None:
        """Sets single selection programmatically."""
        if self.manager and hasattr(self.manager, "selection_helper") and self.manager.selection_helper:
            self.manager.selection_helper.set_selection(prim_path)

    # Class-level reference to the Level Navigator Window
    _navigator_window = None

    # --- Navigation Request & UI Popup ---
    def request_navigate(self, target_level_id: str, item_id: Optional[str] = None, **extra_data) -> bool:
        """Requests LevelManager to switch to target_level_id."""
        if self.manager:
            return self.manager.navigate_to(target_level_id, item_id, **extra_data)
        return False

    def show_level_navigator(self, title: str = "Level Navigator", width: int = 320, height: int = 220) -> None:
        """
        Displays an interactive Level Navigator UI popup.
        Dynamically detects all registered levels in the project and provides navigation buttons.
        Can be called in on_enter() of any level handler.
        """
        try:
            import omni.ui as ui
        except ImportError:
            return

        try:
            if not BaseLevelHandler._navigator_window:
                BaseLevelHandler._navigator_window = ui.Window(title, width=width, height=height)
            else:
                BaseLevelHandler._navigator_window.title = title
                BaseLevelHandler._navigator_window.visible = True

            levels = self.manager.get_registered_levels() if self.manager else {}
            active_id = self.level_id

            with BaseLevelHandler._navigator_window.frame:
                with ui.VStack(spacing=8):
                    ui.Label("Modular Level Navigator", height=24)
                    ui.Label("Klik tombol di bawah untuk berganti level:", height=18)
                    ui.Line(height=2)
                    for lvl_id, disp_name in levels.items():
                        is_active = (lvl_id == active_id)
                        btn = ui.Button(f"{disp_name}" + (" (Current)" if is_active else ""), height=34)
                        btn.enabled = not is_active
                        btn.set_clicked_fn(lambda l=lvl_id: self.request_navigate(l))
        except Exception as e:
            print(f"[{self.level_id}] Level Navigator UI note: {e}")

    def hide_level_navigator(self) -> None:
        """Hides the level navigator popup."""
        if BaseLevelHandler._navigator_window:
            BaseLevelHandler._navigator_window.visible = False

    # --- Telemetry & Web Bridge Hooks ---
    def get_telemetry_payload(self) -> Dict[str, Any]:
        """
        Override this method to expose level-specific metrics and data to the Web Dashboard.
        Automatically embedded in GET /api/telemetry under level_telemetry.data.
        """
        return {}

    def on_telemetry_received(self, data: Dict[str, Any]) -> None:
        """
        Override this method to react to incoming telemetry and commands from Web Dashboard.
        Executes safely on the main Omniverse thread.
        """
        pass

    def apply_cached_telemetry_if_any(self) -> None:
        """Applies cached telemetry if data was received while this level was inactive."""
        if self.manager and hasattr(self.manager, "telemetry_helper") and self.manager.telemetry_helper:
            cached_data = self.manager.telemetry_helper.get_cached_telemetry(self.level_id)
            if cached_data:
                try:
                    self.on_telemetry_received(cached_data)
                except Exception as e:
                    print(f"[{self.level_id}] Note applying cached telemetry on enter: {e}")

    # --- Lifecycle Hooks ---
    @abstractmethod
    def on_enter(self, stage: Usd.Stage, context: NavigationContext) -> None:
        """Called immediately when entering this level and USD stage is ready."""
        pass

    @abstractmethod
    def on_exit(self, stage: Usd.Stage, next_level_id: str) -> Dict[str, Any]:
        """Called when leaving this level. Must clean up overrides and return state dict."""
        pass

    def on_update(self, stage: Usd.Stage, dt: float) -> None:
        """Called every frame during update stream (e.g. 60 FPS)."""
        pass

    def on_selection_changed(
        self,
        stage: Optional[Usd.Stage],
        prim_path: Optional[str],
        zoom_distance: Optional[float] = None,
        distance: Optional[float] = None,
        zoom: Optional[bool] = None
    ) -> None:
        """
        Called whenever user selects or unselects an object in the 3D viewport.

        Args:
            stage: Active USD stage.
            prim_path: Prim path of the single selected object, or None if cleared.
            zoom_distance: Zoom distance (0.0 = stationary camera/no zoom, > 0.0 = smooth zoom to distance).
            distance: Backward-compatible alias for zoom_distance.
            zoom: If False, forces zoom_distance = 0.0 (stationary).
        """
        if not stage:
            return
        if prim_path:
            # Determine zoom distance priority:
            if zoom_distance is not None:
                actual_dist = float(zoom_distance)
            elif distance is not None:
                actual_dist = float(distance)
            elif zoom is False:
                actual_dist = 0.0
            else:
                actual_dist = float(getattr(self, "click_zoom_distance", 0.0))

            self.set_camera_pivot_to_object(stage, prim_path, zoom_distance=actual_dist)
        else:
            print(f"[{self.level_id}] Selection cleared -> Resetting orbit pivot")
            self.reset_camera_pivot(stage)

    # --- Delegated Metadata & Property Inspection ---
    @staticmethod
    def _serialize_usd_value(val: Any) -> Any:
        """Serializes complex OpenUSD/Gf/Sdf/Vt types to JSON-safe primitives."""
        if val is None:
            return None
        if isinstance(val, (int, float, bool, str)):
            return val
        if isinstance(val, (list, tuple)):
            return [BaseLevelHandler._serialize_usd_value(x) for x in val]
        if isinstance(val, dict):
            return {str(k): BaseLevelHandler._serialize_usd_value(v) for k, v in val.items()}
        if hasattr(val, "pathString"):
            return val.pathString
        if hasattr(val, "__iter__"):
            try:
                return [BaseLevelHandler._serialize_usd_value(x) for x in list(val)]
            except Exception:
                pass
        return str(val)

    def get_prim_metadata(self, stage: Optional[Usd.Stage], prim_path: str) -> Dict[str, Any]:
        """
        Extracts comprehensive metadata, material bindings, custom attributes,
        and geometric dimensions of a target prim in a JSON-serializable format.
        """
        if not stage or not prim_path:
            return {}

        prim = stage.GetPrimAtPath(prim_path)
        if not prim or not prim.IsValid():
            return {"error": f"Prim '{prim_path}' is not valid on stage."}

        # 1. Identity & Prim Info
        info: Dict[str, Any] = {
            "name": prim.GetName(),
            "path": prim.GetPath().pathString,
            "type": prim.GetTypeName() or "Xform",
            "is_active": prim.IsActive(),
            "is_defined": prim.IsDefined(),
        }

        # 2. Bound Material (UsdShade) with Smart Recursive Search
        try:
            from pxr import UsdShade, UsdGeom
            binding_api = UsdShade.MaterialBindingAPI(prim)
            mat, _ = binding_api.ComputeBoundMaterial()
            if mat:
                mat_prim = mat.GetPrim()
                info["material"] = {
                    "material_name": mat_prim.GetName(),
                    "material_path": mat_prim.GetPath().pathString,
                    "binding_level": "direct (assigned to this prim)"
                }
            else:
                # 🔍 Smart Recursive Search: Check all child / grandchild meshes
                found_materials = []
                descendant_count = 0
                for descendant in Usd.PrimRange(prim):
                    if descendant == prim:
                        continue
                    descendant_count += 1
                    desc_binding = UsdShade.MaterialBindingAPI(descendant)
                    d_mat, _ = desc_binding.ComputeBoundMaterial()
                    if d_mat:
                        d_prim = d_mat.GetPrim()
                        entry = {
                            "material_name": d_prim.GetName(),
                            "material_path": d_prim.GetPath().pathString,
                            "bound_to": descendant.GetPath().pathString
                        }
                        if entry not in found_materials:
                            found_materials.append(entry)

                if found_materials:
                    if len(found_materials) == 1:
                        info["material"] = {
                            "material_name": found_materials[0]["material_name"],
                            "material_path": found_materials[0]["material_path"],
                            "binding_level": f"recursive (found on descendant: {found_materials[0]['bound_to']})"
                        }
                    else:
                        info["material"] = {
                            "material_name": f"{len(found_materials)} materials in descendants",
                            "binding_level": "multiple descendants",
                            "details": found_materials
                        }
                else:
                    info["material"] = None
                    if descendant_count > 0:
                        info["material_search_status"] = f"No material bound on this prim or any of its {descendant_count} descendants"
        except Exception as e:
            info["material"] = f"Error: {e}"

        # 3. Geometric Bounding Box & Dimensions
        try:
            from pxr import UsdGeom
            bbox_cache = UsdGeom.BBoxCache(Usd.TimeCode.Default(), [
                UsdGeom.Tokens.default_, UsdGeom.Tokens.render, UsdGeom.Tokens.proxy
            ])
            bound = bbox_cache.ComputeWorldBound(prim)
            rng = bound.ComputeAlignedRange()
            if not rng.IsEmpty():
                min_pt = rng.GetMin()
                max_pt = rng.GetMax()
                size = max_pt - min_pt
                info["bounds"] = {
                    "min": [round(min_pt[0], 2), round(min_pt[1], 2), round(min_pt[2], 2)],
                    "max": [round(max_pt[0], 2), round(max_pt[1], 2), round(max_pt[2], 2)],
                    "dimensions": [round(size[0], 2), round(size[1], 2), round(size[2], 2)],
                }
        except Exception:
            pass

        # 4. OpenUSD Prim Metadata (customData, kind, documentation, etc.)
        try:
            raw_meta = prim.GetAllMetadata()
            clean_meta = {}
            for k, v in raw_meta.items():
                clean_meta[k] = self._serialize_usd_value(v)
            info["usd_metadata"] = clean_meta
        except Exception as e:
            info["usd_metadata"] = f"Error: {e}"

        # 5. Custom Data Dictionary
        try:
            custom_data = prim.GetCustomData()
            if custom_data:
                info["custom_data"] = self._serialize_usd_value(custom_data)
        except Exception:
            pass

        # 6. Attributes
        try:
            attributes = {}
            for attr in prim.GetAttributes():
                val = attr.Get()
                attributes[attr.GetName()] = self._serialize_usd_value(val)
            info["attributes"] = attributes
        except Exception:
            pass

        # 7. Children Prims
        try:
            info["children"] = [child.GetName() for child in prim.GetChildren()]
        except Exception:
            pass

        return info
